# Secret patterns for check-secrets.sh — one extended regex per line.
#
# This file IS scanned -- by a dedicated pass in check-secrets.sh, on every run. It is excluded
# only from the diff/tree/untracked passes, so it cannot match its own definitions there.
# If a pattern you add legitimately matches its own text, append `# nosecret` to that line
# rather than exempting the file. Never inline these patterns into the script.
#
# Two tiers. DENY patterns below fail the build. The WARN tier at the end is deliberately
# lower-confidence -- the P1-00 acceptance criterion says "patterns matching ghp_, sk-ant-",
# while the deny patterns require realistic lengths so ordinary prose does not trip them. A
# truncated, redacted-but-recoverable, or short test token sits in the gap. Warn tier prints
# and does not fail (second adversarial review, minor 3).
#
# Required by the P1-00 acceptance criteria: ghp_, sk-ant-, CFBD_API_KEY, and hardcoded
# database connection strings.

# GitHub personal access tokens and friends
gh[pousr]_[A-Za-z0-9]{36,}
github_pat_[A-Za-z0-9_]{22,}

# Anthropic API keys
sk-ant-[A-Za-z0-9_-]{16,}

# Generic provider keys assigned a literal value (not a ${VAR} or env lookup)
CFBD_API_KEY[[:space:]]*[:=][[:space:]]*["']?[A-Za-z0-9/+_-]{12,}
(AWS_SECRET_ACCESS_KEY|ANTHROPIC_API_KEY)[[:space:]]*[:=][[:space:]]*["']?[A-Za-z0-9/+_-]{12,}

# Database connection strings carrying embedded credentials
(postgres|postgresql|mysql|mongodb)(\+srv)?://[^:/[:space:]]+:[^@[:space:]]+@
(Server|Data Source)=[^;]+;.*(Password|Pwd)=[^;[:space:]]+

# Private key material
-----BEGIN [A-Z ]*PRIVATE KEY-----

# --- WARN TIER ---------------------------------------------------------------------------
# Everything below this marker prints a warning and does NOT fail the build. Keep it for
# shapes that are too weak to gate on but worth a human glance.
#WARN#
gh[pousr]_[A-Za-z0-9]{8,35}[^A-Za-z0-9]
sk-ant-[A-Za-z0-9_-]{4,15}[^A-Za-z0-9_-]
